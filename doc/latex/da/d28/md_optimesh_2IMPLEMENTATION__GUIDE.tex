\item Residual = M⁻¹ @ (D @ F) computed exactly (not approximated)

\end{DoxyItemize}

\DoxyCodeLine{\ \ \ \ else:}
\DoxyCodeLine{\ \ \ \ \ \ \ \ abs\_r\ =\ |r\_ij|}
